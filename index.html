// Cập nhật hàm Đăng ký để lưu vào Firestore async function handleAuth() { const email = document.getElementById('email').value; const pass = document.getElementById('password').value; const name = document.getElementById('fullName').value; const repass = document.getElementById('repassword').value; try { if (isLoginMode) { const res = await auth.signInWithEmailAndPassword(email, pass); // MẸO: Nếu nhân viên cũ đăng nhập, tự động cập nhật họ vào danh sách users nếu chưa có await db.collection("users").doc(res.user.uid).set({ ho_ten: res.user.displayName || res.user.email, email: res.user.email, last_login: new Date().toISOString() }, { merge: true }); } else { if(!name || pass !== repass) return alert("Vui lòng kiểm tra lại thông tin!"); const res = await auth.createUserWithEmailAndPassword(email, pass); await res.user.updateProfile({ displayName: name }); // Lưu vào danh bạ users await db.collection("users").doc(res.user.uid).set({ ho_ten: name, email: email, ngay_tao: new Date().toISOString() }); alert("Đăng ký thành công!"); location.reload(); } } catch (e) { alert("Lỗi: " + e.message); } }
